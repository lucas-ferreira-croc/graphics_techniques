function(buildTechnique TECHNIQUE_NAME)
	SET(TECHNIQUEFOLDER ${CMAKE_CURRENT_SOURCE_DIR}/${TECHNIQUE_NAME})
	message(STATUS "Generating project file for technique in ${TECHNIQUEFOLDER}")
	# Main
	file(GLOB SOURCE *.cpp ${BASE_HEADERS} ${TECHNIQUEFOLDER}/*.cpp)
	SET(MAIN_CPP ${TECHNIQUEFOLDER}/${TECHNIQUE_NAME}.cpp)
	if(EXISTS ${TECHNIQUEFOLDER}/main.cpp)
		SET(MAIN_CPP ${TECHNIQUEFOLDER}/main.cpp)
	ENDIF()
	if(EXISTS ${TECHNIQUEFOLDER}/${TECHNIQUE_NAME}.h)
		SET(MAIN_HEADER ${TECHNIQUEFOLDER}/${TECHNIQUE_NAME}.h)
	ENDIF()
	# imgui example requires additional source files
	IF(${TECHNIQUE_NAME} STREQUAL "imgui")
		file(GLOB ADD_SOURCE "../external/imgui/*.cpp")
		SET(SOURCE ${SOURCE} ${ADD_SOURCE})
	ENDIF()
	# Add shaders
	set(SHADER_DIR_GLSL "../shaders/glsl/${TECHNIQUE_NAME}")
	file(GLOB SHADERS_GLSL "${SHADER_DIR_GLSL}/*.vert" "${SHADER_DIR_GLSL}/*.frag" "${SHADER_DIR_GLSL}/*.comp" "${SHADER_DIR_GLSL}/*.geom" "${SHADER_DIR_GLSL}/*.tesc" "${SHADER_DIR_GLSL}/*.tese" "${SHADER_DIR_GLSL}/*.mesh" "${SHADER_DIR_GLSL}/*.task" "${SHADER_DIR_GLSL}/*.rgen" "${SHADER_DIR_GLSL}/*.rchit" "${SHADER_DIR_GLSL}/*.rmiss" "${SHADER_DIR_GLSL}/*.rcall" "${SHADER_DIR_GLSL}/*.rahit" "${SHADER_DIR_GLSL}/*.rint" "${SHADER_DIR_GLSL}/*.glsl")
	set(SHADER_DIR_HLSL "../shaders/hlsl/${TECHNIQUE_NAME}")
	file(GLOB SHADERS_HLSL "${SHADER_DIR_HLSL}/*.vert" "${SHADER_DIR_HLSL}/*.frag" "${SHADER_DIR_HLSL}/*.comp" "${SHADER_DIR_HLSL}/*.geom" "${SHADER_DIR_HLSL}/*.tesc" "${SHADER_DIR_HLSL}/*.tese" "${SHADER_DIR_HLSL}/*.mesh" "${SHADER_DIR_HLSL}/*.task" "${SHADER_DIR_HLSL}/*.rgen" "${SHADER_DIR_HLSL}/*.rchit" "${SHADER_DIR_HLSL}/*.rmiss" "${SHADER_DIR_HLSL}/*.rcall" "${SHADER_DIR_HLSL}/*.rahit" "${SHADER_DIR_HLSL}/*.rint")
	set(SHADER_DIR_SLANG "../shaders/slang/${TECHNIQUE_NAME}")
	file(GLOB SHADERS_SLANG "${SHADER_DIR_SLANG}/*.slang")
	source_group("Shaders\\GLSL" FILES ${SHADERS_GLSL})
	source_group("Shaders\\HLSL" FILES ${SHADERS_HLSL})
	source_group("Shaders\\SLANG" FILES ${SHADERS_SLANG})
	# Add optional readme / tutorial
	file(GLOB README_FILES "${TECHNIQUEFOLDER}/*.md")
	if(WIN32)
		add_executable(${TECHNIQUE_NAME} WIN32 ${MAIN_CPP} ${SOURCE} ${MAIN_HEADER} ${SHADERS_GLSL} ${SHADERS_HLSL} ${SHADERS_SLANG} ${README_FILES})
		target_link_libraries(${TECHNIQUE_NAME} base ${Vulkan_LIBRARY} ${WINLIBS})
	else(WIN32)
		add_executable(${TECHNIQUE_NAME} ${MAIN_CPP} ${SOURCE} ${MAIN_HEADER} ${SHADERS_GLSL} ${SHADERS_HLSL} ${SHADERS_SLANG} ${README_FILES})
		target_link_libraries(${TECHNIQUE_NAME} base )
	endif(WIN32)

	file(MAKE_DIRECTORY ${CMAKE_SOURCE_DIR}/bin)
	set_target_properties(${TECHNIQUE_NAME} PROPERTIES VS_DEBUGGER_WORKING_DIRECTORY ${CMAKE_SOURCE_DIR}/bin)
	if(${TECHNIQUE_NAME} STREQUAL "texture3d")
		find_package(OpenMP)
		if(OpenMP_CXX_FOUND)
			set_target_properties(${TECHNIQUE_NAME} PROPERTIES COMPILE_FLAGS ${OpenMP_CXX_FLAGS})
			target_include_directories(${TECHNIQUE_NAME} PRIVATE ${OpenMP_CXX_INCLUDE_DIRS})
			target_link_libraries(${TECHNIQUE_NAME} ${OpenMP_CXX_LIBRARIES})
		endif()
	endif()

	if(RESOURCE_INSTALL_DIR)
		install(TARGETS ${TECHNIQUE_NAME} DESTINATION ${CMAKE_INSTALL_BINDIR})
	endif()
endfunction(buildTechnique)

function(buildTechniques)
	foreach(TECHNIQUE ${TECHNIQUES})
		buildTechnique(${TECHNIQUE})
	endforeach(TECHNIQUE)
endfunction(buildTechniques)

set(TECHNIQUES
    trianglevulkan13
)

buildTechniques()
